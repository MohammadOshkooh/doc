#RULE

CREATE RULE add_user
AS ON INSERT TO users 
WHERE NEW.username ilike 'a%'
DO ALSO INSERT INTO admins_table(id, username) VALUES (NEW.id , NEW.username);

\d <table_name> -> show rule

SELECT * FROM pg_rules;

#triggers

CREATE OR REPLACE FUNCTION a_tags()
RETURNS trigger 
LANGUAGE plpgsql 
AS $$
BEGIN 
   IF lower(substring(NEW.username FROM 1 FOR 1)) = 'a' THEN
   INSERT INTO admins_table(id, username) VALUES (NEW.id, NEW.username);
   END IF;
   RETURN NEW;
END;
$$;


CREATE OR REPLACE TRIGGER add_a_user BEFORE INSERT ON users FOR EACH ROW 
EXECUTE FUNCTION a_tags();
